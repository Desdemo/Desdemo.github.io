为什么需要重试

最常见的来源于 网络波动或者数据库乐观锁更新失败，重试的时候需要带幂等策略或者版本号

接口取消策略

1 简单重试

func SimpleRetry(attempts int, fn func() error) error {
    var err error
    for i := 0; i < attempts; i++ {
        err = fn()
        if err == nil {
            return nil
        }
    }
    return err
}

2 等待时重试

func DelayRetry(attempts int, fn func() error) error {
	var err error
	for i := 0; i < attempts; i++ {
		err = fn()
		if err == nil {
			return nil
		}
		time.Sleep(50 * time.Millisecond)
	}
	return err
}

3 指数避让重试

为了避免瞬间流量冲击，通常建议等待时间逐渐变长（例如：1秒 -> 2秒 -> 4秒）

package retryutil

import (
	"context"
	"errors"
	"math/rand/v2"
	"time"
)

// RetryConfig 重试配置
type RetryConfig struct {
	Attempts    int           // 最大尝试次数，默认 3
	BaseDelay   time.Duration // 基础延迟，默认 100ms
	MaxDelay    time.Duration // 最大延迟上限，默认 2s（仅指数退避有效）
	Strategy    DelayStrategy // 延迟策略：Fixed / Exponential / FullJitter
	RetryableFn func(error) bool // 错误分类：返回 true=可重试，false=直接返回
	OnRetryFn   func(attempt int, err error) // 每次重试前的回调（可为 nil）
}

// DelayStrategy 延迟策略类型
type DelayStrategy int

const (
	Fixed        DelayStrategy = iota // 固定延迟，每次等待 BaseDelay
	Exponential                       // 指数退避: BaseDelay * 2^(attempt-1)，上限 MaxDelay
	FullJitter                        // 指数退避 + 全抖动: [0, min(MaxDelay, BaseDelay*2^(attempt-1))]
)

// DefaultConfig 返回默认配置：指数退避 + 全抖动，3次，适用于 API 请求
func DefaultConfig() RetryConfig {
	return RetryConfig{
		Attempts:  3,
		BaseDelay: 100 * time.Millisecond,
		MaxDelay:  2 * time.Second,
		Strategy:  FullJitter,
	}
}

// OptimisticLockConfig 返回乐观锁配置：固定短延迟，默认 3 次
func OptimisticLockConfig() RetryConfig {
	return RetryConfig{
		Attempts:  3,
		BaseDelay: 50 * time.Millisecond,
		Strategy:  Fixed,
		RetryableFn: func(err error) bool {
			return errors.Is(err, ErrVersionExpired)
		},
	}
}

// ErrVersionExpired 乐观锁版本冲突错误
var ErrVersionExpired = errors.New("version expired")

// Do 重试执行无返回值的函数
func Do(ctx context.Context, cfg RetryConfig, fn func() error) error {
	_, err := DoWithData(ctx, cfg, func() (any, error) {
		return nil, fn()
	})
	return err
}

// DoWithData 重试执行有返回值的函数
func DoWithData[T any](ctx context.Context, cfg RetryConfig, fn func() (T, error)) (T, error) {
	// 填充默认值
	if cfg.Attempts <= 0 {
		cfg.Attempts = 3
	}
	if cfg.BaseDelay <= 0 {
		cfg.BaseDelay = 100 * time.Millisecond
	}
	if cfg.MaxDelay <= 0 {
		cfg.MaxDelay = 2 * time.Second
	}

	var lastErr error
	for attempt := 0; attempt < cfg.Attempts; attempt++ {
		// 检查上下文是否已取消
		if err := ctx.Err(); err != nil {
			return zero[T](), err
		}

		// 执行
		result, err := fn()

		// 成功，返回
		if err == nil {
			return result, nil
		}
		lastErr = err

		// 判断是否应继续重试
		if cfg.RetryableFn != nil && !cfg.RetryableFn(err) {
			return zero[T](), err
		}

		// 如果是最后一次尝试，不再等待，直接返回错误
		if attempt == cfg.Attempts-1 {
			break
		}

		// 计算延迟
		delay := calcDelay(cfg, attempt+1)

		// 回调通知
		if cfg.OnRetryFn != nil {
			cfg.OnRetryFn(attempt+1, err)
		}

		// 等待，同时监听 context 取消
		select {
		case <-ctx.Done():
			return zero[T](), ctx.Err()
		case <-time.After(delay):
		}
	}

	return zero[T](), lastErr
}

func calcDelay(cfg RetryConfig, attempt int) time.Duration {
	switch cfg.Strategy {
	case Fixed:
		return cfg.BaseDelay
	case Exponential:
		d := cfg.BaseDelay << (attempt - 1) // * 2^(attempt-1)
		if d > cfg.MaxDelay || d < cfg.BaseDelay {
			return cfg.MaxDelay
		}
		return d
	case FullJitter:
		d := cfg.BaseDelay << (attempt - 1)
		if d > cfg.MaxDelay || d < cfg.BaseDelay {
			d = cfg.MaxDelay
		}
		return time.Duration(rand.Int64N(int64(d)))
	default:
		return cfg.BaseDelay
	}
}

func zero[T any]() T {
	var v T
	return v
}
